import { getBonusImageUrl, getRankImageUrl } from "./config";
import { STACK_ROWS } from "./stack";
import type { BonusTier, GameMode, Ninja, PlayerResultInfo, StackPlayer } from "./types";

export interface RoundInput {
  mode: GameMode;
  players: readonly Ninja[];
  counts: readonly number[];
  strikeTimes: readonly (number | undefined)[];
  /** Shinobi Stack towers, indexed by player id. */
  stacks?: readonly Pick<StackPlayer, "rows" | "status" | "finishedAtMs">[];
}

export interface RoundResults {
  winnerIds: ReadonlySet<number>;
  byPlayer: Record<number, PlayerResultInfo>;
}

const SUFFIXES = ["ST", "ND", "RD"] as const;
const BONUS_TIERS_4: readonly BonusTier[] = ["plus100", "plus50", "plus10", "plus0"];
const BONUS_TIERS_3: readonly BonusTier[] = ["plus100", "plus50", "plus0"];

export function getRankInfo(rank: number, playerCount = 4) {
  const bonusTiers = playerCount === 3 ? BONUS_TIERS_3 : BONUS_TIERS_4;
  const bonusTier = bonusTiers[rank - 1];
  return {
    label: `${rank}${SUFFIXES[rank - 1] ?? "TH"} PLACE`,
    rankImage: getRankImageUrl(rank),
    bonusImage: bonusTier ? getBonusImageUrl(bonusTier) : undefined,
  };
}

function competitionRank<T>(items: readonly T[], item: T, isBetter: (a: T, b: T) => boolean) {
  return 1 + items.filter((other) => isBetter(other, item)).length;
}

function placed(rank: number, scoreLabel: string, playerCount: number): PlayerResultInfo {
  const info = getRankInfo(rank, playerCount);
  return {
    rank,
    rankLabel: info.label,
    rankImage: info.rankImage,
    bonusImage: info.bonusImage,
    scoreLabel,
  };
}

function failed(rankLabel: string, scoreLabel: string): PlayerResultInfo {
  return { rank: null, rankLabel, rankImage: getRankImageUrl(null), scoreLabel };
}

export function computeResults({
  mode,
  players,
  counts,
  strikeTimes,
  stacks = [],
}: RoundInput): RoundResults {
  const byPlayer: Record<number, PlayerResultInfo> = {};
  const winnerIds = new Set<number>();
  const playerCount = players.length;

  if (mode === "mash") {
    const entries = players.map((p) => ({ id: p.id, score: counts[p.id] ?? 0 }));
    const noContest = entries.every((e) => e.score === 0);

    for (const entry of entries) {
      const label = `${entry.score} ${entry.score === 1 ? "Strike" : "Strikes"}`;
      if (noContest) {
        byPlayer[entry.id] = failed("NO STRIKES", label);
        continue;
      }
      const rank = competitionRank(entries, entry, (a, b) => a.score > b.score);
      byPlayer[entry.id] = placed(rank, label, playerCount);
      if (rank === 1) winnerIds.add(entry.id);
    }
    return { winnerIds, byPlayer };
  }

  if (mode === "stack") {
    const entries = players.map((p) => {
      const tower = stacks[p.id];
      return {
        id: p.id,
        height: tower?.rows.length ?? 0,
        summit: tower?.status === "summit",
        timeMs: tower?.finishedAtMs ?? 0,
      };
    });
    const noContest = entries.every((e) => e.height === 0);
    // Summiting beats not summiting; summiters rank by speed, the rest by height.
    const isBetter = (a: (typeof entries)[number], b: (typeof entries)[number]) => {
      if (a.summit !== b.summit) return a.summit;
      return a.summit ? a.timeMs < b.timeMs : a.height > b.height;
    };

    for (const entry of entries) {
      const label = entry.summit
        ? `Summit in ${(entry.timeMs / 1000).toFixed(2)}s`
        : `Height ${entry.height}/${STACK_ROWS}`;
      if (noContest) {
        byPlayer[entry.id] = failed("NO BLOCKS", label);
        continue;
      }
      const rank = competitionRank(entries, entry, isBetter);
      byPlayer[entry.id] = placed(rank, label, playerCount);
      if (rank === 1) winnerIds.add(entry.id);
    }
    return { winnerIds, byPlayer };
  }

  const entries = players.map((p) => ({ id: p.id, time: strikeTimes[p.id] }));
  const struck = entries.filter((e): e is { id: number; time: number } => e.time !== undefined);

  for (const entry of entries) {
    if (entry.time === undefined) {
      byPlayer[entry.id] = failed("FAIL", "No Strike");
      continue;
    }
    const rank = competitionRank(struck, { id: entry.id, time: entry.time }, (a, b) => a.time < b.time);
    byPlayer[entry.id] = placed(rank, `${entry.time.toFixed(3)}s left`, playerCount);
    if (rank === 1) winnerIds.add(entry.id);
  }
  return { winnerIds, byPlayer };
}

import { describe, expect, it } from "vitest";
import { DEFAULT_NINJAS } from "./ninjas";
import { computeResults } from "./results";

const players = DEFAULT_NINJAS;

describe("computeResults (mash)", () => {
  it("ranks by score and shares rank on ties", () => {
    const r = computeResults({ mode: "mash", players, counts: [10, 10, 4, 0], strikeTimes: [] });
    expect([0, 1, 2, 3].map((i) => r.byPlayer[i].rank)).toEqual([1, 1, 3, 4]);
    expect([...r.winnerIds]).toEqual([0, 1]);
  });

  it("declares no winner when nobody struck", () => {
    const r = computeResults({ mode: "mash", players, counts: [0, 0, 0, 0], strikeTimes: [] });
    expect(r.winnerIds.size).toBe(0);
    expect(r.byPlayer[0].rank).toBeNull();
    expect(r.byPlayer[0].bonusImage).toBeUndefined();
  });

  it("assigns bonus tiers correctly for 4 players", () => {
    const r = computeResults({ mode: "mash", players, counts: [10, 8, 6, 4], strikeTimes: [] });
    expect(r.byPlayer[0].bonusImage).toContain("plus100");
    expect(r.byPlayer[1].bonusImage).toContain("plus50");
    expect(r.byPlayer[2].bonusImage).toContain("plus10");
    expect(r.byPlayer[3].bonusImage).toContain("plus0");
  });

  it("assigns bonus tiers correctly for 3 players (3rd receives plus0 placeholder)", () => {
    const players3 = players.slice(0, 3);
    const r = computeResults({ mode: "mash", players: players3, counts: [10, 8, 6], strikeTimes: [] });
    expect(r.byPlayer[0].bonusImage).toContain("plus100");
    expect(r.byPlayer[1].bonusImage).toContain("plus50");
    expect(r.byPlayer[2].bonusImage).toContain("plus0");
  });
});

import type { MovingBlock, Row, SpeedTier, StackPlayer } from "./types";

/* ---------- Board geometry (grid units) ---------- */
export const STACK_COLS = 8;
export const STACK_ROWS = 11;
export const START_WIDTH = 3;

/* ---------- Speed curve: slow start, ramps every row ---------- */
/** Slowest tick interval (ms), used for the first row. */
export const MAX_SPEED_MS = 180;
/** Fastest tick interval (ms). */
export const MIN_SPEED_MS = 32;
const SPEED_RAMP = (MAX_SPEED_MS - MIN_SPEED_MS) / (STACK_ROWS - 1);

export const MAX_ACTIVE_BURSTS = 5;
/** Upper bound on moves processed per tick if the tab was throttled. */
export const MAX_CATCH_UP_MOVES = 4;

export const SPEED_TIERS: Record<SpeedTier, { en: string }> = {
  walk: { en: "WALK" },
  run: { en: "RUN" },
  wind: { en: "WIND" },
  lightning: { en: "LIGHTNING" },
};

/** Rank title indexed by number of rows stacked. */
const RANKS = [
  "GENIN", "GENIN", "GENIN",
  "CHUNIN", "CHUNIN", "CHUNIN",
  "JONIN", "JONIN", "JONIN",
  "ANBU",
  "KAGE", "KAGE",
];

/** A fresh moving block at the left edge. */
export function newMovingBlock(width: number = START_WIDTH): MovingBlock {
  return { x: 0, width, dir: 1 };
}

export function createStackPlayer(): StackPlayer {
  return {
    rows: [],
    current: newMovingBlock(),
    status: "climbing",
    nextMoveAt: 0,
    finishedAtMs: null,
    bursts: [],
  };
}

/** Tick interval (ms) for a tower that is `rowCount` rows high. */
export function speedForRows(rowCount: number): number {
  return Math.max(MIN_SPEED_MS, Math.round(MAX_SPEED_MS - rowCount * SPEED_RAMP));
}

/** Current speed as a 0–100 percentage of the full speed range. */
export function speedPercent(rowCount: number): number {
  const ms = speedForRows(rowCount);
  return Math.round(((MAX_SPEED_MS - ms) / (MAX_SPEED_MS - MIN_SPEED_MS)) * 100);
}

export function speedTier(percent: number): SpeedTier {
  if (percent < 20) return "walk";
  if (percent < 50) return "run";
  if (percent < 80) return "wind";
  return "lightning";
}

export function rankForRows(rowCount: number): string {
  return RANKS[Math.min(rowCount, RANKS.length - 1)];
}

/** Move the block one cell, bouncing off the board edges. */
export function advanceBlock(block: MovingBlock): MovingBlock {
  let x = block.x + block.dir;
  let dir = block.dir;
  if (x + block.width > STACK_COLS) {
    x = STACK_COLS - block.width;
    dir = -1;
  }
  if (x < 0) {
    x = 0;
    dir = 1;
  }
  return { ...block, x, dir };
}

/** The part of `block` that rests on `support`, or `null` if they don't overlap. */
export function overlapWith(block: Row, support: Row): Row | null {
  const left = Math.max(block.x, support.x);
  const right = Math.min(block.x + block.width, support.x + support.width);
  const width = right - left;
  return width > 0 ? { x: left, width } : null;
}

/** Deterministic hash → [0, 1). Keeps reducers and effects pure (no Math.random). */
export function unit(seed: number): number {
  let t = (seed + 0x6d2b79f5) | 0;
  t = Math.imul(t ^ (t >>> 15), t | 1);
  t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
  return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
}

const NINJA_BASES: readonly NinjaBase[] = [
  { id: 0, name: "Player 1", key: "A", code: "KeyA", colorName: "red" },
  { id: 1, name: "Player 2", key: "F", code: "KeyF", colorName: "green" },
  { id: 2, name: "Player 3", key: "H", code: "KeyH", colorName: "blue" },
  { id: 3, name: "Player 4", key: "L", code: "KeyL", colorName: "yellow" },
];

export const DEFAULT_NINJAS: Ninja[] = NINJA_BASES.map((base) =>
  withColor({ ...base, color: "", glow: "", bgImage: "" }, base.colorName)
);

export function assignColor(
  ninjas: readonly Ninja[],
  playerId: number,
  colorName: NinjaColor
): Ninja[] {
  const target = ninjas.find((n) => n.id === playerId);
  if (!target || target.colorName === colorName) return [...ninjas];
  const holder = ninjas.find((n) => n.colorName === colorName);

  return ninjas.map((n) => {
    if (n.id === playerId) return withColor(n, colorName);
    if (holder && n.id === holder.id) return withColor(n, target.colorName);
    return n;
  });
}

function createHit(seq: number): Hit {
  return {
    id: seq,
    x: 15 + unit(seq * 3 + 1) * 70,
    y: 20 + unit(seq * 3 + 2) * 50,
    rot: unit(seq * 3 + 3) * 360,
  };
}

export interface GameState {
  phase: Phase;
  mode: GameMode;
  numPlayers: PlayerCount;
  ninjas: Ninja[];
  countdown: number;
  startedAt: number | null;
  counts: number[];
  strikeTimes: (number | undefined)[];
  pulses: number[];
  hits: Hit[][];
  /** Shinobi Stack towers, one per seat. */
  stacks: StackPlayer[];
  hitSeq: number;
  videosDone: boolean[];
  resultsShownAt: number;
}

export type GameAction =

  | { type: "select_player_count"; count: PlayerCount }
  | { type: "select_color"; playerId: number; color: NinjaColor }

  | { type: "countdown_tick" }
  | { type: "begin_play"; at: number }
  | { type: "time_up" }

  | { type: "video_timeout" }
  | { type: "show_results"; at: number }
  | { type: "hit_done"; playerId: number; hitId: number }
  | { type: "stack_tick"; at: number }
  | { type: "burst_done"; playerId: number; burstId: number }
  | { type: "return_to_menu" }
  | { type: "key_down"; code: string; at: number };

const CONFIRM_CODES = new Set(["Enter", "NumpadEnter", "Space"]);

const filled = <T,>(value: T) => Array.from({ length: MAX_PLAYERS }, () => value);

function roundState() {
  return {
    startedAt: null,
    counts: filled(0),
    strikeTimes: filled<number | undefined>(undefined),
    pulses: filled(0),
    hits: Array.from({ length: MAX_PLAYERS }, (): Hit[] => []),
    stacks: Array.from({ length: MAX_PLAYERS }, createStackPlayer),
    videosDone: filled(false),
    resultsShownAt: 0,
  };
}

export function createInitialState(): GameState {
  return {
    phase: "menu",
    mode: "mash",
    numPlayers: 4,
    ninjas: DEFAULT_NINJAS,
    countdown: COUNTDOWN_START,
    hitSeq: 0,
    ...roundState(),
  };
}

function replaceAt<T>(list: readonly T[], index: number, update: (value: T) => T): T[] {
  return list.map((value, i) => (i === index ? update(value) : value));
}

function startRound(state: GameState): GameState {
  // Kunai Bullseye runs its own rounds, countdown and results inside its component.
  if (state.mode === "kunai") return { ...state, ...roundState(), phase: "kunai" };
  return { ...state, ...roundState(), phase: "countdown", countdown: COUNTDOWN_START };
}

function endRound(state: GameState): GameState {
  // Stack rounds skip the win/lose videos so players can study their towers.
  if (state.mode === "stack") return { ...state, phase: "ended" };
  return {
    ...state,
    phase: "video",
    videosDone: Array.from({ length: MAX_PLAYERS }, (_, i) => i >= state.numPlayers),
  };
}

/** Locks the climbing block of `id` onto its tower (or topples the tower). */
function dropBlock(state: GameState, id: number, at: number, elapsed: number): GameState {
  const player = state.stacks[id];
  if (!player || player.status !== "climbing") return state;

  // The first block always lands; later blocks are trimmed to the overlap.
  const top = player.rows[player.rows.length - 1];
  const placed = top ? overlapWith(player.current, top) : { x: player.current.x, width: player.current.width };

  const hitSeq = state.hitSeq + 1;
  let next: StackPlayer;
  if (!placed) {
    next = { ...player, status: "fallen", finishedAtMs: elapsed };
  } else {
    const rows = [...player.rows, placed];
    const bursts = [
      ...player.bursts.slice(-(MAX_ACTIVE_BURSTS - 1)),
      { id: hitSeq, cx: placed.x + placed.width / 2, level: rows.length - 1 },
    ];
    next =
      rows.length >= STACK_ROWS
        ? { ...player, rows, bursts, status: "summit", finishedAtMs: elapsed }
        : {
            ...player,
            rows,
            bursts,
            current: newMovingBlock(placed.width),
            nextMoveAt: at + speedForRows(rows.length),
          };
  }

  const stacks = replaceAt(state.stacks, id, () => next);
  const updated: GameState = {
    ...state,
    hitSeq,
    stacks,
    pulses: replaceAt(state.pulses, id, (n) => n + 1),
  };
  const everyoneDone = state.ninjas
    .slice(0, state.numPlayers)
    .every((p) => stacks[p.id].status !== "climbing");
  return everyoneDone ? endRound(updated) : updated;
}

/** Advances every climbing player's moving block that is due. */
function advanceStacks(state: GameState, at: number): GameState {
  if (state.phase !== "playing" || state.mode !== "stack") return state;

  let changed = false;
  const stacks = state.stacks.map((p, i) => {
    if (i >= state.numPlayers || p.status !== "climbing" || at < p.nextMoveAt) return p;
    let { current, nextMoveAt } = p;
    const interval = speedForRows(p.rows.length);
    for (let n = 0; n < MAX_CATCH_UP_MOVES && at >= nextMoveAt; n++) {
      current = advanceBlock(current);
      nextMoveAt += interval;
    }
    if (at >= nextMoveAt) nextMoveAt = at + interval; // resync after a long stall
    changed = true;
    return { ...p, current, nextMoveAt };
  });
  return changed ? { ...state, stacks } : state;
}

/** When the next stack block is due to move (`null` if nothing is moving). */
export function nextStackDue(state: GameState): number | null {
  if (state.phase !== "playing" || state.mode !== "stack") return null;
  let due: number | null = null;
  for (const p of state.stacks.slice(0, state.numPlayers)) {
    if (p.status === "climbing" && (due === null || p.nextMoveAt < due)) due = p.nextMoveAt;
  }
  return due;
}

function strike(state: GameState, code: string, at: number): GameState {
  if (state.startedAt === null) return state;

  const players = state.ninjas.slice(0, state.numPlayers);
  const player = players.find((n) => n.code === code);
  if (!player) return state;

  const elapsed = Math.max(0, at - state.startedAt);
  if (elapsed >= ROUND_MS[state.mode]) return state;
  if (state.mode === "stack") return dropBlock(state, player.id, at, elapsed);

  const id = player.id;
  if (state.mode === "blind" && state.strikeTimes[id] !== undefined) return state;

  const hitSeq = state.hitSeq + 1;
  const next: GameState = {
    ...state,
    hitSeq,
    pulses: replaceAt(state.pulses, id, (n) => n + 1),
    hits: replaceAt(state.hits, id, (list) => [
      ...list.slice(-(MAX_VISIBLE_HITS - 1)),
      createHit(hitSeq),
    ]),
  };

  if (state.mode === "mash") {
    return { ...next, counts: replaceAt(state.counts, id, (n) => n + 1) };
  }

  const strikeTimes = replaceAt(state.strikeTimes, id, () => (ROUND_MS.blind - elapsed) / 1000);
  const updated = { ...next, strikeTimes };
  return players.every((p) => strikeTimes[p.id] !== undefined) ? endRound(updated) : updated;
}

function handleKey(state: GameState, code: string, at: number): GameState {
  if (code === "Escape") {
    return state.phase === "menu" ? state : { ...state, phase: "menu" };
  }

  switch (state.phase) {
    case "menu": {
      if (CONFIRM_CODES.has(code)) return startRound(state);
      const count = PLAYER_COUNT_CODES[code];
      return count ? { ...state, numPlayers: count } : state;
    }
    case "playing":
      return strike(state, code, at);
    case "results":
      return CONFIRM_CODES.has(code) && at - state.resultsShownAt >= RESULTS_INPUT_LOCK_MS
        ? { ...state, phase: "menu" }
        : state;
    default:
      return state;
  }
}

export function gameReducer(state: GameState, action: GameAction): GameState {
  switch (action.type) {
    case "select_mode":
      return state.phase === "menu" ? { ...state, mode: action.mode } : state;
    case "select_player_count":
      return state.phase === "menu" ? { ...state, numPlayers: action.count } : state;
    case "select_color":
      return state.phase === "menu"
        ? { ...state, ninjas: assignColor(state.ninjas, action.playerId, action.color) }
        : state;
    case "start":
      return state.phase === "menu" ? startRound(state) : state;
    case "countdown_tick":
      return state.phase === "countdown" && state.countdown > 0
        ? { ...state, countdown: state.countdown - 1 }
        : state;
    case "begin_play":
      return state.phase === "countdown" && state.countdown === 0
        ? {
            ...state,
            phase: "playing",
            startedAt: action.at,
            stacks: state.stacks.map((p) => ({ ...p, nextMoveAt: action.at + speedForRows(0) })),
          }
        : state;
    case "time_up":
      return state.phase === "playing" ? endRound(state) : state;
    case "video_ended": {
      if (state.phase !== "video") return state;
      const videosDone = replaceAt(state.videosDone, action.playerId, () => true);
      return videosDone.every(Boolean)
        ? { ...state, videosDone, phase: "ended" }
        : { ...state, videosDone };
    }
    case "video_timeout":
      return state.phase === "video" ? { ...state, phase: "ended" } : state;
    case "show_results":
      return state.phase === "ended"
        ? { ...state, phase: "results", resultsShownAt: action.at }
        : state;
    case "hit_done": {
      const list = state.hits[action.playerId];
      if (!list?.some((h) => h.id === action.hitId)) return state;
      return {
        ...state,
        hits: replaceAt(state.hits, action.playerId, (l) => l.filter((h) => h.id !== action.hitId)),
      };
    }
    case "stack_tick":
      return advanceStacks(state, action.at);
    case "burst_done": {
      const stack = state.stacks[action.playerId];
      if (!stack?.bursts.some((b) => b.id === action.burstId)) return state;
      return {
        ...state,
        stacks: replaceAt(state.stacks, action.playerId, (p) => ({
          ...p,
          bursts: p.bursts.filter((b) => b.id !== action.burstId),
        })),
      };
    }
    case "return_to_menu":
      return state.phase === "results" || state.phase === "kunai" ? { ...state, phase: "menu" } : state;
    case "key_down":
      return handleKey(state, action.code, action.at);
  }
}

import { describe, expect, it } from "vitest";
import {
  MAX_SPEED_MS,
  MIN_SPEED_MS,
  STACK_COLS,
  advanceBlock,
  overlapWith,
  rankForRows,
  speedForRows,
  speedTier,
} from "./stack";

describe("stack logic", () => {
  it("bounces the moving block off both edges", () => {
    const right = advanceBlock({ x: STACK_COLS - 3, width: 3, dir: 1 });
    expect(right).toEqual({ x: STACK_COLS - 3, width: 3, dir: -1 });
    expect(advanceBlock({ x: 0, width: 3, dir: -1 })).toEqual({ x: 0, width: 3, dir: 1 });
    expect(advanceBlock({ x: 2, width: 3, dir: 1 }).x).toBe(3);
  });

  it("trims to the overlap and returns null when there is none", () => {
    expect(overlapWith({ x: 2, width: 3 }, { x: 0, width: 3 })).toEqual({ x: 2, width: 1 });
    expect(overlapWith({ x: 3, width: 2 }, { x: 0, width: 3 })).toBeNull();
  });

  it("speeds up as the tower grows, within bounds", () => {
    expect(speedForRows(0)).toBe(MAX_SPEED_MS);
    expect(speedForRows(100)).toBe(MIN_SPEED_MS);
    expect(speedForRows(5)).toBeLessThan(speedForRows(1));
  });

  it("maps speed and height to tiers and ranks", () => {
    expect(speedTier(0)).toBe("walk");
    expect(speedTier(100)).toBe("lightning");
    expect(rankForRows(0)).toBe("GENIN");
    expect(rankForRows(11)).toBe("KAGE");
    expect(rankForRows(99)).toBe("KAGE");
  });
});
