import { memo, useEffect, useRef, type CSSProperties } from "react";
import { getRowImageUrl } from "../game/config";
import { STACK_COLS, STACK_ROWS } from "../game/stack";
import type { Row, StackPlayer } from "../game/types";
import { SakuraStorm, ShurikenBurst } from "./StackEffects";

interface StackFieldProps {
  playerId: number;
  stack: StackPlayer;
  /** Increments on every drop; replays the landing flash. */
  pulseId: number;
  onBurstDone: (playerId: number, burstId: number) => void;
}

/** Warm the cache so a block never shows up without its art. */
for (const width of [1, 2, 3]) {
  new Image().src = getRowImageUrl(width);
}

/** Positions a block as percentages so the field scales with its container. */
const blockStyle = ({ x, width }: Row, level: number): CSSProperties =>
  ({
    left: `${(x / STACK_COLS) * 100}%`,
    bottom: `${(level / STACK_ROWS) * 100}%`,
    width: `${(width / STACK_COLS) * 100}%`,
    height: `${100 / STACK_ROWS}%`,
    "--n": width,
  }) as CSSProperties;

/** Transparent row art stretched over the block; the player color shows through. */
const BlockArt = ({ width }: { width: number }) => (
  <img className="block-art" src={getRowImageUrl(width)} alt="" draggable={false} />
);

/** One player's tower: grid, placed blocks, the moving block and effects. */
export const StackField = memo(function StackField({
  playerId,
  stack,
  pulseId,
  onBurstDone,
}: StackFieldProps) {
  const { rows, current, status, bursts } = stack;
  const top = rows[rows.length - 1];
  const climbing = status === "climbing";
  const fieldRef = useRef<HTMLDivElement>(null);

  // Replay the landing flash on every drop without remounting the field.
  useEffect(() => {
    const field = fieldRef.current;
    if (!field || pulseId === 0) return;
    field.classList.remove("flash");
    void field.getBoundingClientRect(); // force reflow so the animation restarts
    field.classList.add("flash");
  }, [pulseId]);

  return (
    <div className="stack-wrap">
      <div ref={fieldRef} className={`stack-field is-${status}`}>
        {top && climbing && <div className="stack-block guide" style={blockStyle(top, rows.length)} />}

        {rows.map((row, level) => (
          <div key={level} className="stack-block placed" style={blockStyle(row, level)}>
            <BlockArt width={row.width} />
          </div>
        ))}

        {climbing && (
          <div className="stack-block moving" style={blockStyle(current, rows.length)}>
            <BlockArt width={current.width} />
            <div className="blade-sheen" />
          </div>
        )}

        {status === "fallen" && (
          <div className="stack-block missed" style={blockStyle(current, rows.length)}>
            <BlockArt width={current.width} />
          </div>
        )}

        {bursts.map((burst) => (
          <ShurikenBurst key={burst.id} burst={burst} playerId={playerId} onDone={onBurstDone} />
        ))}

        {status === "summit" && <SakuraStorm />}

        {!climbing && (
          <div className={`stack-overlay ${status}`}>
            <div className="stack-overlay-title">{status === "summit" ? "SUMMIT!" : "FALLEN"}</div>
          </div>
        )}
      </div>
    </div>
  );
});

import { memo, type CSSProperties } from "react";
import { unit } from "../game/random";
import { STACK_COLS, STACK_ROWS } from "../game/stack";
import type { Burst } from "../game/types";

const PARTICLES = 10;

interface ShurikenBurstProps {
  burst: Burst;
  playerId: number;
  onDone: (playerId: number, burstId: number) => void;
}

/** Shuriken burst played each time a block lands. Removes itself when finished. */
export const ShurikenBurst = memo(function ShurikenBurst({ burst, playerId, onDone }: ShurikenBurstProps) {
  const stars = Array.from({ length: PARTICLES }, (_, i) => {
    const seed = burst.id * 31 + i * 7;
    const angle = (i / PARTICLES) * Math.PI * 2 + unit(seed) * 0.5;
    const dist = 26 + unit(seed + 1) * 34;
    return {
      id: i,
      dx: Math.cos(angle) * dist,
      dy: Math.sin(angle) * dist * 0.7 - 10,
      size: 8 + unit(seed + 2) * 6,
      tone: i % 3 === 0 ? "#e8e8f0" : i % 2 ? "var(--c)" : "var(--a)",
      spin: (360 + unit(seed + 3) * 540) * (i % 2 ? 1 : -1),
      delay: unit(seed + 4) * 50,
    };
  });

  return (
    <div
      className="stack-burst"
      style={{
        left: `${(burst.cx / STACK_COLS) * 100}%`,
        top: `${((STACK_ROWS - burst.level - 0.5) / STACK_ROWS) * 100}%`,
      }}
      onAnimationEnd={(e) => {
        // Only the container's own lifetime animation ends the burst.
        if (e.target === e.currentTarget) onDone(playerId, burst.id);
      }}
    >
      <div className="stack-burst-ring" />
      {stars.map((s) => (
        <div
          key={s.id}
          className="stack-shuriken-bit"
          style={
            {
              width: s.size,
              height: s.size,
              background: s.tone,
              "--dx": `${s.dx}px`,
              "--dy": `${s.dy}px`,
              "--spin": `${s.spin}deg`,
              animationDelay: `${s.delay}ms`,
            } as CSSProperties
          }
        />
      ))}
    </div>
  );
});

import { memo, useEffect, useRef, useState, type CSSProperties } from "react";
import { getOutcomeImageUrl } from "../game/config";
import type { Ninja, Hit, GameMode, Outcome, PlayerResultInfo, StackPlayer } from "../game/types";
import { Shuriken } from "./Shuriken";
import { StackField } from "./StackField";
import { StackStats } from "./StackStats";

interface PlayerCardProps {
  player: Ninja;
  mode: GameMode;
  count: number;
  maxCount: number;
  isLeading: boolean;
  pulseId: number;
  hits: readonly Hit[];
  /** Tower state, used in Shinobi Stack mode. */
  stack?: StackPlayer;
  lockedTime?: number;
  isLocked: boolean;
  outcome?: Outcome;
  videoUrl?: string;
  result?: PlayerResultInfo;
  onVideoEnd: (playerId: number) => void;
  onHitDone: (playerId: number, hitId: number) => void;
  onBurstDone: (playerId: number, burstId: number) => void;
}

export const PlayerCard = memo(function PlayerCard({
  player,
  mode,
  count,
  maxCount,
  isLeading,
  pulseId,
  hits,
  stack,
  lockedTime,
  isLocked,
  outcome,
  videoUrl,
  result,
  onVideoEnd,
  onHitDone,
  onBurstDone,
}: PlayerCardProps) {
  const videoRef = useRef<HTMLVideoElement>(null);
  const [failedVideoUrl, setFailedVideoUrl] = useState<string>();

  const activeVideoUrl = videoUrl && videoUrl !== failedVideoUrl ? videoUrl : undefined;

  useEffect(() => {
    const video = videoRef.current;
    if (!activeVideoUrl || !video) return;
    video.currentTime = 0;
    video.play().catch((error: unknown) => {
      if (error instanceof DOMException && error.name === "NotAllowedError") {
        onVideoEnd(player.id);
      }
    });
  }, [activeVideoUrl, onVideoEnd, player.id]);

  const handleVideoError = () => {
    setFailedVideoUrl(videoUrl);
    onVideoEnd(player.id);
  };

  const bgImage = outcome ? getOutcomeImageUrl(player.colorName, outcome) : player.bgImage;

  const isStack = mode === "stack" && stack !== undefined;

  const resultBadge = result && (
    <div className={`card-result-badge ${isStack ? "compact" : ""}`}>
      <img src={result.rankImage} alt={result.rankLabel} className="card-rank-img" />
      <div className="card-result-score">{result.scoreLabel}</div>
      {result.bonusImage && <img src={result.bonusImage} alt="" className="card-bonus-img" />}
    </div>
  );

  return (
    <div
      className={`card mode-${mode} ${isLeading ? "leading" : ""} ${isLocked ? "card-locked" : ""} ${
        outcome ? `outcome-${outcome}` : ""
      } ${activeVideoUrl ? "has-video" : ""}`}
      style={{ "--c": player.color, "--g": player.glow } as CSSProperties}
    >
      {activeVideoUrl ? (
        <video
          ref={videoRef}
          key={activeVideoUrl}
          src={activeVideoUrl}
          autoPlay
          playsInline
          muted
          preload="auto"
          className="card-outcome-video"
          onEnded={() => onVideoEnd(player.id)}
          onError={handleVideoError}
        />
      ) : (
        <>
          <div className="card-bg-image" style={{ backgroundImage: `url("${bgImage}")` }} />
          <div className="card-overlay" />
        </>
      )}

      <div className="card-name">{player.name}</div>

      {isStack && stack ? (
        <>
          <StackField
            playerId={player.id}
            stack={stack}
            pulseId={pulseId}
            onBurstDone={onBurstDone}
          />
          {resultBadge ?? <StackStats rows={stack.rows} />}
        </>
      ) : result ? (
        resultBadge
      ) : activeVideoUrl ? null : (
        <>
          <div className="shuriken-zone">
            <Shuriken pulseId={pulseId} />
            {hits.map((h) => (
              <div key={h.id} className="hit" style={{ left: `${h.x}%`, top: `${h.y}%` }}>
                <div className="slash" style={{ transform: `rotate(${h.rot}deg)` }} />
                <span className="plus" onAnimationEnd={() => onHitDone(player.id, h.id)}>
                  {mode === "blind" ? "LOCKED" : "+1"}
                </span>
              </div>
            ))}
          </div>

          <div className="count" key={`c${count}`}>
            {mode === "mash" ? (
              count
            ) : isLocked ? (
              <span className="locked-badge">LOCKED IN</span>
            ) : (
              <span className="waiting-badge">WAITING</span>
            )}
          </div>

          {mode === "mash" && (
            <div className="power">
              <div className="power-fill" style={{ height: `${(count / maxCount) * 100}%` }} />
            </div>
          )}

          {mode === "blind" && lockedTime !== undefined && (
            <div className="locked-time-preview">Struck!</div>
          )}
        </>
      )}

      {!activeVideoUrl && !result && (
        <kbd className={`big-key ${isLocked ? "key-disabled" : ""}`}>{player.key}</kbd>
      )}
    </div>
  );
});

import { memo, useSyncExternalStore } from "react";
import { BLIND_THRESHOLD_MS } from "../game/config";
import type { Phase, GameMode, Clock } from "../game/types";

interface TimerProps {
  phase: Phase;
  mode: GameMode;
  clock: Clock;
  /** Full round length, used for the idle display and the progress bar. */
  durationMs: number;
}

const formatSeconds = (ms: number) => (ms / 1000).toFixed(2);

export const Timer = memo(function Timer({ phase, mode, clock, durationMs }: TimerProps) {
  const timeLeft = useSyncExternalStore(clock.subscribe, clock.getSnapshot);

  const isPlaying = phase === "playing";
  const isBlindHidden = mode === "blind" && isPlaying && timeLeft <= BLIND_THRESHOLD_MS;
  const isPostRound = phase === "video" || phase === "ended" || phase === "results";
  const showLive = isPlaying || isPostRound;

  const displayTime = isBlindHidden
    ? "??.?? (BLIND)"
    : formatSeconds(showLive ? timeLeft : durationMs);
  const widthPct = isBlindHidden ? 0 : showLive ? (timeLeft / durationMs) * 100 : 100;
  const isDanger = !isBlindHidden && timeLeft < 3000;

  return (
    <div className="timer-wrap">
      <div className={`timer-text ${isBlindHidden ? "blind-hidden" : ""}`}>{displayTime}</div>
      <div className="timer-bar">
        <div
          className={`timer-fill ${isDanger ? "danger" : ""} ${isBlindHidden ? "timer-hidden" : ""}`}
          style={{ width: `${widthPct}%` }}
        />
      </div>
    </div>
  );
});
