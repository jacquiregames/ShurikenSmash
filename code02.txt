import { useEffect, useMemo, useRef, type CSSProperties } from "react";
import { BAR_CM, BULLSEYE_D, ZONES } from "../game/config";
import { usePlayers } from "../players";
import type { Phase, Run } from "../game/useEngine";

interface Props {
  index: number;
  run: Run;
  phase: Phase;
  setMarkerEl: (i: number, el: HTMLElement | null) => void;
  onPress: (i: number) => void;
}

export default function AimLane({ index, run, phase, setMarkerEl, onPress }: Props) {
  const p = usePlayers()[index];
  const padRef = useRef<HTMLButtonElement | null>(null);

  const armed = phase === "live" && run.status === "aiming";
  const locked = run.status === "flying" || run.status === "landed";

  /* Physical key feedback on the mechanical cap */
  useEffect(() => {
    const down = (e: KeyboardEvent) => {
      if (e.repeat || e.key.toLowerCase() !== p.key) return;
      padRef.current?.classList.add("down");
    };
    const up = (e: KeyboardEvent) => {
      if (e.key.toLowerCase() !== p.key) return;
      padRef.current?.classList.remove("down");
    };
    window.addEventListener("keydown", down);
    window.addEventListener("keyup", up);
    return () => {
      window.removeEventListener("keydown", down);
      window.removeEventListener("keyup", up);
    };
  }, [p.key]);

  const ticks = useMemo(() => {
    const out: number[] = [];
    for (let cm = -BAR_CM; cm <= BAR_CM; cm += 4) out.push(50 + (cm / BAR_CM) * 45);
    return out;
  }, []);

  const status =
    run.status === "landed"
      ? run.bull
        ? "🎯 BULLSEYE!"
        : `${ZONES[run.zone].label} (+${run.points})`
      : run.status === "flying"
        ? "⚡ In Flight..."
        : armed
          ? "PRESS TO THROW!"
          : phase === "countdown"
            ? "STEADY..."
            : "STAND BY";

  const cls = [
    "lane",
    armed ? "is-active" : "",
    locked ? "locked" : "",
    phase === "countdown" ? "waiting" : "",
    run.bull && locked ? "bull" : "",
  ]
    .filter(Boolean)
    .join(" ");

  return (
    <div
      className={cls}
      style={{ ["--c" as string]: p.color } as CSSProperties}
      onPointerDown={() => armed && onPress(index)}
    >
      {/* Player Lane Header */}
      <div className="lane-head">
        <button
          type="button"
          ref={padRef}
          className="pad"
          tabIndex={-1}
          aria-label={`${p.label} throw key ${p.key.toUpperCase()}`}
          title={`Press [${p.key.toUpperCase()}] or tap here to throw`}
          onPointerDown={(e) => {
            e.stopPropagation();
            padRef.current?.classList.add("down");
            onPress(index);
          }}
          onPointerUp={() => padRef.current?.classList.remove("down")}
          onPointerLeave={() => padRef.current?.classList.remove("down")}
        >
          <span className="pad-surface">
            <span className="pad-key">{p.key.toUpperCase()}</span>
          </span>
          <span className="pad-halo" />
        </button>

        <div className="lane-name">
          <div className="t"> 
            <span className="p-nm">{p.name}</span> 
          </div> 
        </div>

        <div className={`lane-status${armed ? " ready" : run.status === "landed" ? " landed" : " idle"}`}>
          {status}
        </div>
      </div>

      {/* Target Aiming Track */}
      <div className="track-container">
        <div className="track">
          {/* Target Zone Ring overlays */}
          <div className="zone-band">
            <span className="zb-wood left" />
            <span className="zb-outer left" />
            <span className="zb-mid left" />
            <span className="zb-inner" />
            <span className="zb-mid right" />
            <span className="zb-outer right" />
            <span className="zb-wood right" />
          </div>

          {/* Grid calibration ticks */}
          {ticks.map((left, i) => (
            <span key={`t${i}`} className="grid-tick" style={{ left: `${left}%` }} />
          ))}
          {ticks.map((left, i) => (
            <span key={`b${i}`} className="grid-tick b" style={{ left: `${left}%` }} />
          ))}

          {/* Central Golden Bullseye Gate */}
          <div className="track-center">
            <span className="tc-notch top" />
            <span className="tc-core" />
            <span className="tc-notch btm" />
          </div>

          {/* Sliding Kunai Reticle Marker */}
          <div
            className="marker"
            ref={(el) => {
              setMarkerEl(index, el);
            }}
          >
            <div className="marker-arrowhead" />
            <div className="marker-trail" />
          </div>

          {/* Flash line on release */}
          {locked && <div className="track-lock-pulse" />}
        </div>

        {/* Ruler scale labels */}
        <div className="track-scale">
          <span>−{BAR_CM}cm</span>
          <span className="scale-mid">🎯 的中 BULLSEYE (±{(BULLSEYE_D * BAR_CM).toFixed(1)}cm)</span>
          <span>+{BAR_CM}cm</span>
        </div>
      </div>

      {/* Live Result Readout */}
      <div className="lane-readout">
        {locked ? (
          <div className="readout-locked">
            <span className="cm">
              {run.bull ? "🎯 0.0" : run.cm.toFixed(1)}
              <small>cm</small>
            </span>
            <span className="zone-pill" style={{ borderColor: ZONES[run.zone].color, color: ZONES[run.zone].color }}>
              {ZONES[run.zone].label}
            </span>
            <span className="pts">
              <b>+{run.points}</b> PTS
            </span>
            {run.auto && <span className="auto-tag">fuse timeout</span>}
          </div>
        ) : (
          <span className="empty">
            {armed ? (
              <span className="prompt-throw">
                PRESS <b className="highlight-key">[{p.key.toUpperCase()}]</b> TO LOOSE KUNAI
              </span>
            ) : (
              "Awaiting throw signal..."
            )}
          </span>
        )}
      </div>
    </div>
  );
}

import { useMemo, type CSSProperties } from "react";
import { ROUNDS, rankWord } from "../game/config";
import { usePlayers } from "../players";
import { Shuriken } from "./Icons";

interface Props {
  winners: number[];
  reason: "bull" | "score";
  history: number[][];
  totals: number[];
  onRematch: () => void;
  onTitle: () => void;
}

export default function GameOver({ winners, history, totals, onRematch, onTitle }: Props) {
  const PLAYERS = usePlayers();
  const order = useMemo(
    () => PLAYERS.map((_, i) => i).sort((a, b) => totals[b] - totals[a] || a - b),
    [PLAYERS, totals],
  );

  const petals = useMemo(
    () =>
      Array.from({ length: 22 }, (_, i) => {
        const r = (n: number) => ((Math.sin(i * 12.9898 + n * 78.233) * 43758.5453) % 1 + 1) % 1;
        return {
          left: `${(r(1) * 100).toFixed(2)}%`,
          dur: `${(5 + r(2) * 6).toFixed(2)}s`,
          delay: `${(-r(3) * 9).toFixed(2)}s`,
          dx: `${((r(4) * 2 - 1) * 260).toFixed(0)}px`,
          spin: `${(r(5) * 1080 - 540).toFixed(0)}deg`,
          size: 8 + r(6) * 14,
          kind: i % 3,
          color: i % 3 === 0 ? "#e3b23c" : i % 3 === 1 ? "#e2483a" : "#f4ead6",
        };
      }),
    [],
  );

  const tie = winners.length > 1;
  const lead = PLAYERS[winners[0]];
  const best = Math.max(...totals);

  return (
    <div className="over" role="dialog" aria-modal="true" aria-label="Match result">
      {petals.map((p, i) => (
        <span
          key={i}
          className="petal"
          style={
            {
              left: p.left,
              animationDuration: p.dur,
              animationDelay: p.delay,
              "--dx": p.dx,
              "--spin": p.spin,
              color: p.color,
            } as CSSProperties
          }
        >
          {p.kind === 2 ? (
            <span style={{ display: "block", width: p.size, height: p.size * 0.7, background: p.color, opacity: 0.5, borderRadius: "60% 10% 60% 10%" }} />
          ) : (
            <span style={{ display: "block", opacity: 0.75 }}>
              <Shuriken size={p.size} />
            </span>
          )}
        </span>
      ))}

      <div className="over-card">
        <div className="seal">忍</div>
        <div className="over-kicker">終局 · match complete</div>
        <h2 className="over-title">
          {tie ? (
            <>
              SHARED <em>HONOUR</em>
            </>
          ) : (
            <>
              {lead.name} <em>WINS</em>
            </>
          )}
        </h2>
        <p className="over-sub">
          {tie
            ? `Level on ${best} points after ${ROUNDS} rounds — a tie is a victory for both`
            : `Highest accuracy after ${ROUNDS} rounds · ${best} of ${ROUNDS * 100} points`}
        </p>

        <div className="winners">
          {winners.map((w, k) => {
            const p = PLAYERS[w];
            return (
              <div
                className="wcard"
                key={p.id}
                style={{ ["--c" as string]: p.color, ["--dl" as string]: `${0.15 + k * 0.12}s` } as CSSProperties}
              >
                <span className="halo" />
                <span className="cap">{p.key.toUpperCase()}</span>
                <div className="nm">{p.name}</div>
                <div className="sc">
                  {totals[w]}
                  <small>PTS</small>
                </div>
                <div className="tag">
                  {tie ? "co-champion · 引き分け" : rankWord(0)} · {p.jp}
                </div>
              </div>
            );
          })}
        </div>

        <table className="table">
          <thead>
            <tr>
              <th style={{ width: 46 }}>#</th>
              <th className="name">Shinobi</th>
              {Array.from({ length: ROUNDS }, (_, r) => (
                <th key={r}>Round {r + 1}</th>
              ))}
              <th>Total</th>
            </tr>
          </thead>
          <tbody>
            {order.map((i, rank) => {
              const p = PLAYERS[i];
              const isW = winners.includes(i);
              return (
                <tr
                  key={p.id}
                  className={`${isW ? "win" : ""} ${rank > 0 && totals[i] === 0 ? "dim" : ""}`}
                  style={{ ["--c" as string]: p.color } as CSSProperties}
                >
                  <td style={{ color: isW ? "var(--gold)" : "var(--ash)" }}>{rank + 1}</td>
                  <td className="name">
                    {p.key.toUpperCase()} · {p.name}
                  </td>
                  {Array.from({ length: ROUNDS }, (_, r) => {
                    const v = history[r]?.[i];
                    return <td key={r}>{v === undefined ? "–" : v}</td>;
                  })}
                  <td className="tot">{totals[i]}</td>
                </tr>
              );
            })}
          </tbody>
        </table>

        <div className="over-actions">
          <button type="button" className="btn" onClick={onRematch}>
            <span className="pulse-ring" />
            Rematch
            <span className="kb">SPACE</span>
          </button>
          <button type="button" className="btn ghost" onClick={onTitle}>
            Back to menu
            <span className="kb">ESC</span>
          </button>
          <span className="title-note">Same keys. Faster markers. Sharper shame.</span>
        </div>
      </div>
    </div>
  );
}

export type Phase = "title" | "countdown" | "live" | "roundEnd" | "gameOver";

export function useEngine(players: readonly KunaiPlayer[]): Engine {

  const [history, setHistory] = useState<number[][]>([]);
  const [totals, setTotals] = useState<number[]>(() => zeros(N));
  const [winners, setWinners] = useState<number[] | null>(null);
  const [winReason, setWinReason] = useState<"bull" | "score" | null>(null);
  const [banner, setBanner] = useState<Banner | null>(null);
  const [callout, setCallout] = useState<{ i: number; throwId: number } | null>(null);
  const [shakeId, setShakeId] = useState(0);
  const [pulseId, setPulseId] = useState(0);
  const [muted, setMuted] = useState(false);

  /* ---- refs that mirror state so the rAF loop never reads stale values ---- */
  const phaseRef = useRef<Phase>("title");
  const roundRef = useRef(1);
  const runsRef = useRef<Run[]>(freshRuns(N));
  const historyRef = useRef<number[][]>([]);
  const posRef = useRef<number[]>([0, 0, 0, 0]);
  const dirRef = useRef<number[]>([1, -1, 1, -1]);
  const lockedRef = useRef<boolean[]>(flags(N));
  const landedRef = useRef<boolean[]>(flags(N));
  const endScheduled = useRef(false);

  const lastSec = useRef(99);
  const seq = useRef(0);

  const fuseFillRef = useRef<HTMLDivElement | null>(null);
  const fuseValRef = useRef<HTMLSpanElement | null>(null);

    setRuns(next);

    timers.current.push(id);
    return id;

    timers.current = [];

    const rn = roundRef.current;

        syncRuns(runsRef.current.map((r, idx) => (idx === i ? { ...r, status: "landed" } : r)));
        sfx.hit(points, bull);
        setPulseId((p) => p + 1);
        setCallout({ i, throwId });
        if (bull || points >= 62) setShakeId((s) => s + 1);

        if (bull) {
          setBanner({
            jp: "的中！",
            title: "BULLSEYE",
            sub: `${players[i].name} — dead centre! +100 PTS`,
            gold: true,
          });
          after(2300, () => setBanner((b) => (b && b.jp === "的中！" ? null : b)));
        }

        after(1900, () => setCallout((c) => (c && c.throwId === throwId ? null : c)));

        if (phaseRef.current === "live" && landedRef.current.every(Boolean)) {
          after(1000, () => endRound());
        }
      });
    },
    // eslint-disable-next-line react-hooks/exhaustive-deps
    [after, endRound, paint, players],
  );

  const press = useCallback(
    (i: number) => {
      if (phaseRef.current !== "live") return;
      if (lockedRef.current[i]) return;
      lock(i, false);
    },
    [lock],
  );

  /* ---------------- the loop ---------------- */

  useEffect(() => {
    let raf = 0;
    let last = performance.now();

    const step = (t: number) => {
      raf = requestAnimationFrame(step);
      const dt = Math.min(0.05, Math.max(0, (t - last) / 1000));
      last = t;

      const p = phaseRef.current;
      if (p !== "countdown" && p !== "live") return;

      const sp = SPEED[roundRef.current - 1] ?? SPEED[0];
      let moving = false;
      for (let i = 0; i < N; i++) {
        if (lockedRef.current[i]) continue;
        moving = true;
        let np = posRef.current[i] + dirRef.current[i] * sp * dt;
        if (np >= 1) {
          np = 2 - np;
          dirRef.current[i] = -1;
        } else if (np <= -1) {
          np = -2 - np;
          dirRef.current[i] = 1;
        }
        posRef.current[i] = np;
        const el = markerEls.current[i];
        if (el) {
          el.style.setProperty("--pos", np.toFixed(4));
          el.style.setProperty("--dir", dirRef.current[i] > 0 ? "1" : "-1");
          el.style.setProperty("--trail-o", (0.3 + Math.min(0.55, sp * 0.28)).toFixed(2));
          el.style.setProperty("--speed-scale", (0.7 + sp * 0.45).toFixed(2));
        }
      }
      if (!moving) return;

      if (p === "live") {
        const total = ROUND_TIME[roundRef.current - 1] ?? 10;
        fuseLeft.current = Math.max(0, fuseLeft.current - dt);
        const rem = fuseLeft.current;
        const frac = Math.max(0, Math.min(1, rem / total));
        if (fuseFillRef.current) fuseFillRef.current.style.setProperty("--f", frac.toFixed(4));
        if (fuseValRef.current) fuseValRef.current.textContent = rem.toFixed(1);
        const danger = rem <= 3.05;
        if (fuseRef.current) fuseRef.current.classList.toggle("is-danger", danger);
        if (fuseValRef.current) fuseValRef.current.classList.toggle("is-danger", danger);

        const sec = Math.ceil(rem);
        if (sec !== lastSec.current) {
          lastSec.current = sec;
          if (sec > 0 && sec <= 3) sfx.fuseTick();
        }

        if (rem <= 0) {
          for (let i = 0; i < N; i++) if (!lockedRef.current[i]) lock(i, true);
        }
      }
    };

    raf = requestAnimationFrame(step);
    return () => cancelAnimationFrame(raf);
  }, [N, lock]);

  /* ---------------- top level actions ---------------- */

  const start = useCallback(() => {
    sfx.unlock();
    setHistory([]);
    historyRef.current = [];
    setTotals(zeros(N));
    beginCountdown(1);
  }, [N, beginCountdown]);

  const restart = useCallback(() => {
    sfx.ui();
    start();
  }, [start]);

  const toTitle = useCallback(() => {
    clearTimers();
    sfx.ui();
    phaseRef.current = "title";
    setPhase("title");
    setBanner(null);
    setCallout(null);
    setWinners(null);
    setWinReason(null);
    setHistory([]);
    historyRef.current = [];
    setTotals(zeros(N));
    setRound(1);
    roundRef.current = 1;
    syncRuns(freshRuns(N));
  }, [N, clearTimers]);

  const toggleMute = useCallback(() => {
    setMuted((m) => {
      sfx.setMuted(!m);
      if (m) sfx.ui();
      return !m;
    });
  }, []);

  useEffect(() => () => clearTimers(), [clearTimers]);

  const maxTotal = Math.max(0, ...totals);
  const leaders = maxTotal > 0 ? totals.map((t, i) => (t === maxTotal ? i : -1)).filter((i) => i >= 0) : [];

  return {
    phase,
    round,
    count,
    runs,
    history,
    totals,
    leaders,
    winners,
    winReason,
    banner,
    callout,
    shakeId,
    pulseId,
    muted,
    speedMult: SPEED[round - 1] / SPEED[0],
    fuseTotal: ROUND_TIME[round - 1],
    setMarkerEl,
    fuseRef,
    fuseFillRef,
    fuseValRef,
    press,
    start,
    restart,
    toTitle,
    toggleMute,
  };
}
