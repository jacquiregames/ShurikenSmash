import type { Ninja, NinjaColor } from "../../game/types";

/* ============================================================
   Tuning, scoring and player roster for Kunai Bullseye.
   The aim bar runs from -1 (far left) to +1 (far right).
   A normalised |pos| of MATO_D is exactly the rim of the target.
   ============================================================ */

/** A seat in the match: your menu's ninja (key + color) plus the Kunai flavour text. */
export interface KunaiPlayer {
  id: number;
  /** Lower-case key the player throws with (e.g. "a"). */
  key: string;
  /** Seat label, "P1".."P4". */
  label: string;
  name: string;
  jp: string;
  role: string;
  color: string;
}

/** Name, kanji and title follow the colour a player picked in the menu. */
const IDENTITY: Record<NinjaColor, { name: string; jp: string; role: string }> = {
  red: { name: "AKEMI", jp: "朱", role: "Crimson Fang" },
  yellow: { name: "FUYU", jp: "金", role: "Gilded Shadow" },
  green: { name: "HAYATE", jp: "翠", role: "Wind Step" },
  blue: { name: "RIN", jp: "蒼", role: "Moon On The Water" },
  orange: { name: "KAEDE", jp: "橙", role: "Autumn Ember" },
  purple: { name: "MURASAKI", jp: "紫", role: "Twilight Veil" },
};

/** Builds the match roster from the ninjas that are playing. */
export function buildKunaiPlayers(ninjas: readonly Ninja[]): KunaiPlayer[] {
  return ninjas.map((n, i) => ({
    id: i,
    key: n.key.toLowerCase(),
    label: `P${i + 1}`,
    color: n.color,
    ...IDENTITY[n.colorName],
  }));
}

export const ROUNDS = 3;

/** marker travel speed (normalised bar-lengths per second) per round */
export const SPEED = [0.78, 1.22, 1.78];

/** seconds on the fuse per round */
export const ROUND_TIME = [11, 9.5, 8];

/** |pos| that lands exactly on the target rim */
export const MATO_D = 0.45;

/** anything tighter than this splits the gold dot = 100 points */
export const BULLSEYE_D = 0.032;

/** ring radii, expressed in bar units (they match the drawn mato exactly) */
export const R_INNER = 0.095;
export const R_MID = 0.19;

/** half-length of the bar in centimetres, for flavour readouts */
export const BAR_CM = 26;

/** beyond this the kunai is stuck in the backboard: no points */
export const REACH = 0.62;

/** flight time of a thrown kunai, ms */
export const FLIGHT_MS = 520;

export type ZoneKey = "bull" | "inner" | "mid" | "outer" | "wood" | "off";

export const ZONES: Record<ZoneKey, { label: string; jp: string; color: string; pts: string }> = {
  bull: { label: "Bullseye", jp: "的中", color: "#e3b23c", pts: "100" },
  inner: { label: "Inner Ring", jp: "内", color: "#f4ead6", pts: "81 – 94" },
  mid: { label: "Mid Ring", jp: "中", color: "#c9603f", pts: "63 – 80" },
  outer: { label: "Outer Ring", jp: "外", color: "#4f7fa8", pts: "20 – 62" },
  wood: { label: "Backboard", jp: "板", color: "#7d6a4a", pts: "1 – 19" },
  off: { label: "Off Target", jp: "外れ", color: "#5d5f6b", pts: "0" },
};

export function zoneFor(d: number): ZoneKey {
  if (d <= BULLSEYE_D) return "bull";
  if (d <= R_INNER) return "inner";
  if (d <= R_MID) return "mid";
  if (d <= MATO_D) return "outer";
  if (d < REACH) return "wood";
  return "off";
}

export function scoreFor(d: number): number {
  if (d <= BULLSEYE_D) return 100;
  if (d >= REACH) return 0;
  const t = (REACH - d) / REACH;
  return Math.max(1, Math.round(100 * Math.pow(t, 1.25)));
}

/** rank wording for the final screen */
export function rankWord(n: number): string {
  return ["CHAMPION", "SECOND", "THIRD", "FOURTH"][n] ?? "";
}

/* WebAudio synthesizer for Kunai Bullseye — 100% procedural, no external audio assets needed */

let ctx: AudioContext | null = null;
let master: GainNode | null = null;
let noiseBuf: AudioBuffer | null = null;
let muted = false;

function ac(): AudioContext | null {
  if (typeof window === "undefined") return null;
  if (!ctx) {
    const AC =
      window.AudioContext ||
      (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext;
    if (!AC) return null;
    ctx = new AC();
    master = ctx.createGain();
    master.gain.value = 0.55;
    master.connect(ctx.destination);
  }
  if (ctx.state === "suspended") void ctx.resume();
  return ctx;
}

export function unlock() {
  ac();
}

export function setMuted(m: boolean) {
  muted = m;
  if (master) master.gain.value = m ? 0 : 0.55;
}

export function isMuted() {
  return muted;
}

function noise(c: AudioContext) {
  if (!noiseBuf) {
    noiseBuf = c.createBuffer(1, c.sampleRate * 1.5, c.sampleRate);
    const d = noiseBuf.getChannelData(0);
    for (let i = 0; i < d.length; i++) d[i] = Math.random() * 2 - 1;
  }
  const s = c.createBufferSource();
  s.buffer = noiseBuf;
  return s;
}

function env(node: AudioNode, t: number, a: number, dur: number, peak: number) {
  const g = ctx!.createGain();
  g.gain.setValueAtTime(0.0001, t);
  g.gain.exponentialRampToValueAtTime(peak, t + a);
  g.gain.exponentialRampToValueAtTime(0.0001, t + dur);
  node.connect(g);
  g.connect(master!);
  return g;
}

function tone(
  type: OscillatorType,
  f0: number,
  f1: number,
  dur: number,
  peak: number,
  delay = 0,
) {
  const c = ac();
  if (!c || muted) return;
  const t = c.currentTime + delay;
  const o = c.createOscillator();
  o.type = type;
  o.frequency.setValueAtTime(f0, t);
  o.frequency.exponentialRampToValueAtTime(Math.max(20, f1), t + dur);
  env(o, t, 0.006, dur, peak);
  o.start(t);
  o.stop(t + dur + 0.06);
}

function burst(dur: number, peak: number, f0: number, f1: number, delay = 0, q = 1) {
  const c = ac();
  if (!c || muted) return;
  const t = c.currentTime + delay;
  const s = noise(c);
  const bp = c.createBiquadFilter();
  bp.type = "bandpass";
  bp.Q.value = q;
  bp.frequency.setValueAtTime(f0, t);
  bp.frequency.exponentialRampToValueAtTime(Math.max(40, f1), t + dur);
  s.connect(bp);
  env(bp, t, 0.005, dur, peak);
  s.start(t);
  s.stop(t + dur + 0.06);
}

/** Marker locked: snappy wooden clamp + katana unsheathe chime */
export function lock() {
  tone("square", 1320, 680, 0.05, 0.2);
  tone("triangle", 1880, 1100, 0.08, 0.15);
  burst(0.06, 0.18, 3800, 1200, 0, 2.5);
}

/** Kunai thrown: whoosh of cutting wind with metallic blade whistle */
export function throwSfx() {
  burst(0.32, 0.28, 480, 3800, 0, 1.1);
  tone("sawtooth", 280, 1150, 0.28, 0.07);
  tone("sine", 600, 1200, 0.22, 0.04);
}

/** Kunai impact into wood/straw: solid thud + vibrating twang oscillation */
export function hit(points: number, bull: boolean) {
  if (bull) {
    gong();
    return;
  }
  const c = ac();
  if (!c || muted) return;

  // Solid wood thud
  const thudPitch = Math.max(65, 140 - points * 0.5);
  tone("sine", thudPitch, 38, 0.28, 0.48);
  burst(0.18, 0.32, 1200, 140, 0, 1.2);

  // Metal resonance / blade twang
  tone("triangle", 440 + points * 3.5, 220, 0.35, 0.14, 0.01);
  tone("sine", 880 + points * 4, 340, 0.22, 0.08, 0.02);

  // Subtle spring twang modulation
  try {
    const t = c.currentTime + 0.02;
    const osc = c.createOscillator();
    osc.type = "sawtooth";
    osc.frequency.setValueAtTime(320 + points * 2, t);
    osc.frequency.exponentialRampToValueAtTime(110, t + 0.3);
    const g = env(osc, t, 0.01, 0.32, 0.08);
    // Vibrato
    const lfo = c.createOscillator();
    lfo.frequency.value = 24;
    const lfoGain = c.createGain();
    lfoGain.gain.value = 18;
    lfo.connect(lfoGain);
    lfoGain.connect(osc.frequency);
    lfo.start(t);
    lfo.stop(t + 0.35);
    osc.start(t);
    osc.stop(t + 0.35);
    g.gain.setValueAtTime(0.08, t);
  } catch {
    // fallback if advanced graph fails
  }
}

/** Bullseye: Majestic Japanese temple gong (Daisō) with golden shimmer */
export function gong() {
  const c = ac();
  if (!c || muted) return;

  // Heavy fundamental and deep sub
  tone("sine", 130, 126, 2.8, 0.6);
  tone("sine", 196, 192, 2.4, 0.45);
  tone("sine", 294, 288, 2.1, 0.35);
  tone("triangle", 392, 384, 1.8, 0.25);
  tone("triangle", 588, 574, 1.5, 0.18, 0.01);
  tone("sine", 880, 860, 1.2, 0.12, 0.02);
  
  // Strike impact burst + high shimmer
  burst(0.65, 0.28, 3200, 420, 0, 0.85);
  burst(0.4, 0.15, 6000, 2200, 0.02, 1.8);
}

/** Countdown: Authentic Taiko drum strikes (ドン、ドン、ドン、カッ！) */
export function count(n: number) {
  if (n === 0) {
    // GO! (カッ - sharp wooden bachi rimshot)
    burst(0.18, 0.42, 2800, 600, 0, 2.2);
    tone("square", 840, 420, 0.22, 0.28);
    tone("triangle", 1120, 560, 0.2, 0.18);
  } else {
    // 3, 2, 1 (ドン - deep Taiko drum beat)
    const base = 85 + (3 - n) * 12;
    tone("sine", base * 1.8, base * 0.6, 0.38, 0.55);
    tone("triangle", base * 2.4, base * 0.8, 0.28, 0.32);
    burst(0.14, 0.35, 850, 120, 0, 1.4);
  }
}

/** Round clear fanfare */
export function roundUp() {
  const notes = [392, 494, 587, 784];
  notes.forEach((f, i) => {
    tone("triangle", f, f * 1.01, 0.32, 0.18, i * 0.08);
    tone("sine", f / 2, f / 2, 0.35, 0.12, i * 0.08);
  });
}

/** Match victory fanfare */
export function victory() {
  const notes = [330, 392, 494, 587, 659, 784, 988, 1175];
  notes.forEach((f, i) => {
    tone("triangle", f, f, 0.45, 0.22, i * 0.1);
    tone("sine", f / 2, f / 2, 0.55, 0.14, i * 0.1);
  });
  setTimeout(() => gong(), 750);
}

import { describe, expect, it } from "vitest";
import { DEFAULT_NINJAS, withColor } from "../../game/ninjas";
import { BULLSEYE_D, MATO_D, REACH, buildKunaiPlayers, scoreFor, zoneFor } from "./config";

describe("kunai scoring", () => {
  it("scores a bullseye 100 and anything past the reach 0", () => {
    expect(scoreFor(0)).toBe(100);
    expect(scoreFor(BULLSEYE_D)).toBe(100);
    expect(scoreFor(REACH)).toBe(0);
    expect(scoreFor(REACH + 0.1)).toBe(0);
  });

  it("scores closer throws higher, never below 1 on the board", () => {
    expect(scoreFor(0.1)).toBeGreaterThan(scoreFor(0.3));
    expect(scoreFor(REACH - 0.001)).toBeGreaterThanOrEqual(1);
  });

  it("maps distance to zones", () => {
    expect(zoneFor(0)).toBe("bull");
    expect(zoneFor(0.05)).toBe("inner");
    expect(zoneFor(0.15)).toBe("mid");
    expect(zoneFor(MATO_D)).toBe("outer");
    expect(zoneFor(0.5)).toBe("wood");
    expect(zoneFor(REACH)).toBe("off");
  });
});

describe("buildKunaiPlayers", () => {
  it("uses the menu keys and colors, one seat per ninja", () => {
    const players = buildKunaiPlayers(DEFAULT_NINJAS.slice(0, 3));
    expect(players).toHaveLength(3);
    expect(players.map((p) => p.key)).toEqual(["a", "f", "h"]);
    expect(players.map((p) => p.label)).toEqual(["P1", "P2", "P3"]);
    expect(players[0].color).toBe(DEFAULT_NINJAS[0].color);
  });

  it("gives each colour its own name so seats stay distinguishable", () => {
    const players = buildKunaiPlayers(
      (["red", "yellow", "green", "blue"] as const).map((c, i) => withColor(DEFAULT_NINJAS[i], c))
    );
    expect(new Set(players.map((p) => p.name)).size).toBe(4);
    const six = buildKunaiPlayers(
      (["red", "yellow", "green", "blue", "orange", "purple"] as const).map((c) => withColor(DEFAULT_NINJAS[0], c))
    );
    expect(new Set(six.map((p) => p.name)).size).toBe(6);
  });
});
